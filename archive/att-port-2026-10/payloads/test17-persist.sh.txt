#!/sbin/sh
# Test17: separate from the RAM collector. Only new diagnostic files on cache.
PATH=/sbin
export PATH
umask 077
SELF=/sbin/axon-diag-persist.sh
RAM=/tmp/axon-diag-test17
MOUNTPOINT=/tmp/axon-cache-test17
DEST=$MOUNTPOINT/axon-diag-test17
NODE=/dev/block/mmcblk0p36
EXPECTED_CACHE_UUID=__AXON_CACHE_UUID__

cache_mount_state() {
    # 0 = exact private ext4 rw mount; 2 = absent; 1 = conflicting/other mount.
    found=2
    while IFS=' ' read -r dev point fs options rest; do
        resolved=
        case "$dev" in /dev/block/*) resolved=$(/sbin/readlink -f "$dev") ;; esac
        if [ "$point" != "$MOUNTPOINT" ]; then
            # Do not mount a second view of cache if another actor owns it.
            [ "$resolved" != "$NODE" ] || return 1
            continue
        fi
        [ "$fs" = ext4 ] || return 1
        case ",$options," in *,rw,*) ;; *) return 1 ;; esac
        [ "$resolved" = "$NODE" ] || return 1
        found=0
    done < /proc/mounts
    return "$found"
}

copy_new() {
    src=$1; name=$2
    [ -f "$src" ] && [ ! -L "$src" ] || return 0
    [ ! -e "$DEST/$name" ] && [ ! -L "$DEST/$name" ] || return 0
    # Destination directory was created exclusively by this worker. Noclobber
    # also protects each file; output is capped independently of the collector.
    /sbin/head -c 262144 "$src" > "$DEST/$name" || return 1
    source_hash=$(/sbin/sha256sum "$src") || return 1
    dest_hash=$(/sbin/sha256sum "$DEST/$name") || return 1
    source_hash=${source_hash%% *}; dest_hash=${dest_hash%% *}
    [ "$source_hash" = "$dest_hash" ] || return 1
    printf '%s  %s\n' "$dest_hash" "$name" >> "$DEST/SHA256SUMS"
}

worker() {
    # Filled by the image builder only after fresh read-only cache verification.
    case "$EXPECTED_CACHE_UUID" in ????????-????-????-????-????????????) ;; *) exit 10 ;; esac
    case "$EXPECTED_CACHE_UUID" in *[!0-9a-f-]*) exit 10 ;; esac
    n=0
    while [ ! -b "$NODE" ]; do
        n=$((n + 1)); [ "$n" -le 10 ] || exit 11
        /sbin/sleep 1
    done
    [ "$(/sbin/cat /sys/class/block/mmcblk0p36/start)" = 15204376 ] || exit 13
    [ "$(/sbin/cat /sys/class/block/mmcblk0p36/size)" = 3356672 ] || exit 14
    identity=$(/sbin/timeout -k 1 3 /sbin/blkid "$NODE") || exit 15
    case "$identity" in *'TYPE="ext4"'*) ;; *) exit 16 ;; esac
    case "$identity" in *"UUID=\"$EXPECTED_CACHE_UUID\""*) ;; *) exit 17 ;; esac
    # Keep global /cache in RAM: exposing real cache there would also let
    # TWRP rotate its pre-existing recovery logs outside our new directory.
    /sbin/mkdir "$MOUNTPOINT" || exit 18
    [ -d "$MOUNTPOINT" ] && [ ! -L "$MOUNTPOINT" ] || exit 18
    cache_mount_state
    result=$?
    if [ "$result" = 2 ]; then
        # Exactly one mount attempt. Normal ext4 journal handling; no fsck,
        # formatting, noload, remounting, or mount over an existing filesystem.
        /sbin/timeout -k 1 5 /sbin/mount -t ext4 \
            -o rw,nosuid,nodev,noexec,noatime,sync "$NODE" "$MOUNTPOINT" || exit 19
    else
        exit 20
    fi
    cache_mount_state || exit 21
    # No mkdir -p and no reuse: a previous diagnostic directory aborts saving.
    /sbin/mkdir "$DEST" || exit 22
    set -C
    printf 'Test17 new-file diagnostics only\ncache=%s\nuuid=%s\n' "$NODE" "$EXPECTED_CACHE_UUID" > "$DEST/ABOUT.txt"
    printf '' > "$DEST/SHA256SUMS"
    n=0
    while [ "$n" -lt 40 ]; do
        n=$((n + 1))
        for snap in 01 02 03; do
            for kind in identity dmesg recovery usb processes stacks; do
                copy_new "$RAM/$snap-$kind.ready" "$snap-$kind.txt" || exit 23
            done
        done
        if [ -f "$RAM/collector.ready" ]; then
            copy_new "$RAM/collector.ready" collector.txt || exit 24
            printf 'All available fixed snapshots copied and hash-checked.\n' > "$DEST/COMPLETE.txt"
            # This can block in the kernel, but only the independent worker.
            /sbin/sync
            exit 0
        fi
        /sbin/sleep 1
    done
    printf 'Collector deadline reached; available files only.\n' > "$DEST/INCOMPLETE.txt"
    /sbin/sync
    exit 25
}

case "$1" in
    --worker) worker ;;
    '')
        # No synchronous wait on a mount or disk write in init or the collector.
        # One worker group only. D-state I/O may survive signals; never retry it.
        /sbin/setsid /sbin/timeout -k 2 55 /sbin/sh "$SELF" --worker &
        pid=$!
        /sbin/sleep 60
        /sbin/kill -TERM -- "-$pid" 2>/dev/null
        /sbin/kill -KILL -- "-$pid" 2>/dev/null
        exit 0
        ;;
    *) exit 2 ;;
esac

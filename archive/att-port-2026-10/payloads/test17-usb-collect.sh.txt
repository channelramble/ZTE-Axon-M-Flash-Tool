#!/sbin/sh
# Test17: RAM-only, finite diagnostics. Never mount or write a block device.
PATH=/sbin
export PATH
umask 077
RAM=/tmp/axon-diag-test17
SELF=/sbin/axon-diag-collect.sh
KINDS='identity dmesg recovery usb processes stacks'

show_file() {
    printf '\n--- %s ---\n' "$1"
    if [ -r "$1" ]; then /sbin/head -c 32768 "$1"; else printf 'unavailable\n'; fi
}

job() {
    case "$1" in
        identity)
            /sbin/uname -a
            show_file /proc/uptime
            show_file /proc/cmdline
            show_file /proc/mounts
            show_file /proc/partitions
            show_file /proc/meminfo
            printf '\n--- properties ---\n'
            /sbin/getprop
            ;;
        dmesg) /sbin/dmesg | /sbin/tail -c 250000 ;;
        recovery)
            printf '%s\n' '--- /tmp/recovery.log (last 250000 bytes) ---'
            if [ -r /tmp/recovery.log ]; then /sbin/tail -c 250000 /tmp/recovery.log; fi
            ;;
        usb)
            for p in /sys/devices/soc/6a00000.ssusb/mode \
                /config/usb_gadget/g1/UDC /config/usb_gadget/g1/idVendor \
                /config/usb_gadget/g1/idProduct \
                /config/usb_gadget/g1/configs/b.1/strings/0x409/configuration \
                /sys/kernel/debug/devices_deferred; do show_file "$p"; done
            printf '\n--- functionfs/configfs/UDC directory state ---\n'
            /sbin/ls -ld /dev/usb-ffs /dev/usb-ffs/adb /dev/usb-ffs/adb/* \
                /config/usb_gadget/g1/configs/b.1/* /sys/class/udc/*
            n=0
            for d in /sys/class/udc/*; do
                [ -e "$d" ] || continue
                n=$((n + 1)); [ "$n" -le 8 ] || break
                /sbin/readlink -f "$d"
                for a in state current_speed maximum_speed function uevent; do show_file "$d/$a"; done
            done
            n=0
            for d in /sys/bus/platform/devices/6a00000.*; do
                [ -e "$d" ] || continue
                n=$((n + 1)); [ "$n" -le 8 ] || break
                /sbin/readlink -f "$d/driver"
                show_file "$d/uevent"
            done
            ;;
        processes)
            /sbin/ps -A -T
            show_file /proc/1/status
            show_file /proc/1/wchan
            show_file /proc/1/stack
            ;;
        stacks)
            captured=0; seen=
            capture_process() {
                p=$1
                case " $seen " in *" $p "*) return 0 ;; esac
                [ -d "$p" ] || return 0
                seen="$seen $p"
                comm=; IFS= read -r comm < "$p/comm" 2>/dev/null || return 0
                for t in "$p"/task/[0-9]*; do
                    [ -d "$t" ] || continue
                    [ "$captured" -lt 96 ] || return 0
                    captured=$((captured + 1))
                    printf '\n=== process %s (%s), thread %s ===\n' "$p" "$comm" "$t"
                    show_file "$t/comm"
                    show_file "$t/status"
                    show_file "$t/wchan"
                    show_file "$t/stack"
                done
            }
            # Services first: a probe-retry storm must not crowd them out.
            priority_count=0
            for pid in 1 $(/sbin/pidof recovery adbd); do
                case "$pid" in ''|*[!0-9]*) continue ;; esac
                priority_count=$((priority_count + 1))
                [ "$priority_count" -le 8 ] || break
                capture_process "/proc/$pid"
            done
            # Digit-length ordering reaches early kernel workers before PID1000+.
            scanned=0
            for p in /proc/[0-9] /proc/[0-9][0-9] /proc/[0-9][0-9][0-9] \
                /proc/[0-9][0-9][0-9][0-9] /proc/[0-9][0-9][0-9][0-9][0-9]; do
                [ -d "$p" ] || continue
                scanned=$((scanned + 1)); [ "$scanned" -le 512 ] || break
                [ "$captured" -lt 96 ] || return 0
                comm=; IFS= read -r comm < "$p/comm" 2>/dev/null || continue
                case "$comm" in init|recovery|adbd|ueventd|kworker*) capture_process "$p" ;; esac
            done
            ;;
        *) return 2 ;;
    esac
}

finish_job() {
    key=$1; result=$2
    [ -f "$RAM/$key.ready" ] && return 0
    # Publish atomically: the persister must never see a half-written ready file.
    # A racing late child cannot overwrite a supervisor's partial snapshot.
    (
        set -C
        final="$RAM/$key.final.$$"
        { printf 'test17 snapshot=%s result=%s\n' "$key" "$result"
          [ ! -f "$RAM/$key.part" ] || /sbin/head -c 250000 "$RAM/$key.part"
        } > "$final" || exit 1
        /sbin/ln "$final" "$RAM/$key.ready"
        /sbin/rm "$final"
    ) 2>/dev/null
}

case "$1" in
    --job) job "$2"; exit $? ;;
    --run)
        case "$2" in 01|02|03) ;; *) exit 2 ;; esac
        case "$3" in identity|dmesg|recovery|usb|processes|stacks) ;; *) exit 2 ;; esac
        [ -d "$RAM" ] && [ ! -L "$RAM" ] || exit 2
        key=$2-$3
        # At most 512 KiB even where shell file-size units are 1024 bytes.
        ulimit -f 512 || exit 2
        set -C
        /sbin/timeout -k 1 4 /sbin/sh "$SELF" --job "$3" > "$RAM/$key.part" 2>&1
        result=$?
        finish_job "$key" "$result"
        exit 0
        ;;
    '') ;;
    *) exit 2 ;;
esac

# Fail closed if this boot's directory already exists; never reuse old output.
/sbin/mkdir "$RAM" || exit 1
set -C
printf 'Test17 collector started; kernel/ramdisk changes are recorded by the host.\n' > "$RAM/collector-status.txt"
for snap in 01 02 03; do
    pids=
    for kind in $KINDS; do
        /sbin/setsid /sbin/sh "$SELF" --run "$snap" "$kind" &
        pids="$pids $!"
    done
    /sbin/sleep 6
    for pid in $pids; do
        # Never wait for a D-state job. There are only 18 launched job groups.
        /sbin/kill -TERM -- "-$pid" 2>/dev/null
        /sbin/kill -KILL -- "-$pid" 2>/dev/null
    done
    for kind in $KINDS; do finish_job "$snap-$kind" abandoned-or-timeout; done
    printf 'snapshot %s finished\n' "$snap" >> "$RAM/collector-status.txt"
    case "$snap" in 01) /sbin/sleep 6 ;; 02) /sbin/sleep 12 ;; esac
done
printf 'Collector finished: nominal snapshots at 0, 12, 30 seconds; 18 jobs maximum.\n' >> "$RAM/collector-status.txt"
/sbin/head -c 4096 "$RAM/collector-status.txt" > "$RAM/collector.final"
/sbin/ln "$RAM/collector.final" "$RAM/collector.ready"
/sbin/rm "$RAM/collector.final"
exit 0

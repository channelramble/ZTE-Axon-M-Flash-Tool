#!/system/bin/sh
# Test22: RAM-only, finite diagnostics. Never mount or write a block device.
PATH=/system/bin
export PATH
umask 077
RAM=/dev/axon-diag-test22
SELF=/system/etc/axon-diag-test20-collect.sh
KINDS='identity dmesg logcat usb processes stacks graphics'

show_file() {
    printf '\n--- %s ---\n' "$1"
    if [ -r "$1" ]; then /system/bin/head -c 32768 "$1"; else printf 'unavailable\n'; fi
}

job() {
    case "$1" in
        identity)
            /system/bin/uname -a
            show_file /proc/uptime
            show_file /proc/cmdline
            show_file /proc/mounts
            show_file /proc/partitions
            show_file /proc/meminfo
            printf '\n--- properties ---\n'
            /system/bin/getprop
            ;;
        dmesg) /system/bin/dmesg | /system/bin/tail -c 250000 ;;
        logcat) /system/bin/logcat -b all -d | /system/bin/tail -c 250000 ;;
        graphics)
            /system/bin/ls -l /dev/kgsl-3d0 /dev/graphics /sys/class/kgsl/kgsl-3d0
            for p in /sys/class/graphics/fb0/blank /sys/class/graphics/fb0/modes \
                /sys/class/kgsl/kgsl-3d0/gpuclk /sys/class/kgsl/kgsl-3d0/pwrlevel \
                /sys/kernel/debug/devices_deferred; do show_file "$p"; done
            /system/bin/dumpsys SurfaceFlinger
            ;;
        usb)
            for p in /sys/devices/soc/6a00000.ssusb/mode \
                /config/usb_gadget/g1/UDC /config/usb_gadget/g1/idVendor \
                /config/usb_gadget/g1/idProduct \
                /config/usb_gadget/g1/configs/b.1/strings/0x409/configuration \
                /sys/kernel/debug/devices_deferred; do show_file "$p"; done
            printf '\n--- functionfs/configfs/UDC directory state ---\n'
            /system/bin/ls -ld /dev/usb-ffs /dev/usb-ffs/adb /dev/usb-ffs/adb/* \
                /config/usb_gadget/g1/configs/b.1/* /sys/class/udc/*
            n=0
            for d in /sys/class/udc/*; do
                [ -e "$d" ] || continue
                n=$((n + 1)); [ "$n" -le 8 ] || break
                /system/bin/readlink -f "$d"
                for a in state current_speed maximum_speed function uevent; do show_file "$d/$a"; done
            done
            n=0
            for d in /sys/bus/platform/devices/6a00000.*; do
                [ -e "$d" ] || continue
                n=$((n + 1)); [ "$n" -le 8 ] || break
                /system/bin/readlink -f "$d/driver"
                show_file "$d/uevent"
            done
            ;;
        processes)
            /system/bin/ps -A -T
            show_file /proc/1/status
            show_file /proc/1/wchan
            show_file /proc/1/stack
            ;;
        stacks)
            captured=0; seen=
            capture_process() {
                p=$1
                case " $seen " in *" $p "*) return 0 ;; esac
                [ -d "$p" ] || return 0
                seen="$seen $p"
                comm=; IFS= read -r comm < "$p/comm" 2>/dev/null || return 0
                for t in "$p"/task/[0-9]*; do
                    [ -d "$t" ] || continue
                    [ "$captured" -lt 96 ] || return 0
                    captured=$((captured + 1))
                    printf '\n=== process %s (%s), thread %s ===\n' "$p" "$comm" "$t"
                    show_file "$t/comm"
                    show_file "$t/status"
                    show_file "$t/wchan"
                    show_file "$t/stack"
                done
            }
            # Services first: a probe-retry storm must not crowd them out.
            priority_count=0
            for pid in 1 $(/system/bin/pidof adbd surfaceflinger zygote zygote64 system_server); do
                case "$pid" in ''|*[!0-9]*) continue ;; esac
                priority_count=$((priority_count + 1))
                [ "$priority_count" -le 8 ] || break
                capture_process "/proc/$pid"
            done
            # Digit-length ordering reaches early kernel workers before PID1000+.
            scanned=0
            for p in /proc/[0-9] /proc/[0-9][0-9] /proc/[0-9][0-9][0-9] \
                /proc/[0-9][0-9][0-9][0-9] /proc/[0-9][0-9][0-9][0-9][0-9]; do
                [ -d "$p" ] || continue
                scanned=$((scanned + 1)); [ "$scanned" -le 512 ] || break
                [ "$captured" -lt 96 ] || return 0
                comm=; IFS= read -r comm < "$p/comm" 2>/dev/null || continue
                case "$comm" in init|adbd|surfaceflinger|zygote*|system_server|ueventd|kworker*) capture_process "$p" ;; esac
            done
            ;;
        *) return 2 ;;
    esac
}

finish_job() {
    key=$1; result=$2
    [ -f "$RAM/$key.ready" ] && return 0
    # Publish atomically: the persister must never see a half-written ready file.
    # A racing late child cannot overwrite a supervisor's partial snapshot.
    (
        set -C
        final="$RAM/$key.final.$$"
        { printf 'test22 snapshot=%s result=%s\n' "$key" "$result"
          [ ! -f "$RAM/$key.part" ] || /system/bin/head -c 250000 "$RAM/$key.part"
        } > "$final" || exit 1
        /system/bin/ln "$final" "$RAM/$key.ready"
        /system/bin/rm "$final"
    ) 2>/dev/null
}

case "$1" in
    --job) job "$2"; exit $? ;;
    --run)
        case "$2" in 01|02|03|04) ;; *) exit 2 ;; esac
        case "$3" in identity|dmesg|logcat|usb|processes|stacks|graphics) ;; *) exit 2 ;; esac
        [ -d "$RAM" ] && [ ! -L "$RAM" ] || exit 2
        key=$2-$3
        # At most 512 KiB even where shell file-size units are 1024 bytes.
        ulimit -f 512 || exit 2
        set -C
        /system/bin/timeout -k 1 4 /system/bin/sh "$SELF" --job "$3" > "$RAM/$key.part" 2>&1
        result=$?
        finish_job "$key" "$result"
        exit 0
        ;;
    '') ;;
    *) exit 2 ;;
esac

# Fail closed if this boot's directory already exists; never reuse old output.
/system/bin/mkdir "$RAM" || exit 1
set -C
printf 'Test22 collector started; kernel/ramdisk changes are recorded by the host.\n' > "$RAM/collector-status.txt"
for snap in 01 02 03 04; do
    pids=
    for kind in $KINDS; do
        /system/bin/setsid /system/bin/sh "$SELF" --run "$snap" "$kind" &
        pids="$pids $!"
    done
    /system/bin/sleep 6
    for pid in $pids; do
        # Never wait for a D-state job. There are only 28 launched job groups.
        /system/bin/kill -TERM -- "-$pid" 2>/dev/null
        /system/bin/kill -KILL -- "-$pid" 2>/dev/null
    done
    for kind in $KINDS; do finish_job "$snap-$kind" abandoned-or-timeout; done
    printf 'snapshot %s finished\n' "$snap" >> "$RAM/collector-status.txt"
    case "$snap" in 01) /system/bin/sleep 9 ;; 02) /system/bin/sleep 24 ;; 03) /system/bin/sleep 69 ;; esac
done
printf 'Collector finished: nominal snapshots at 0, 15, 45, 120 seconds; 28 jobs maximum.\n' >> "$RAM/collector-status.txt"
/system/bin/head -c 4096 "$RAM/collector-status.txt" > "$RAM/collector.final"
/system/bin/ln "$RAM/collector.final" "$RAM/collector.ready"
/system/bin/rm "$RAM/collector.final"
exit 0

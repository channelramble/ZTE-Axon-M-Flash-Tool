#!/system/bin/sh
# Test22: reuse Android's existing cache mount, never mount/format a partition.
PATH=/system/bin
export PATH
umask 077
SELF=/system/etc/axon-diag-test20-persist.sh
RAM=/dev/axon-diag-test22
DEST=/cache/axon-diag-test22
NODE=/dev/block/mmcblk0p36
EXPECTED_CACHE_UUID=__PRIVATE_UUID__

cache_mount_state() {
    # Require precisely the expected writable cache mount, with no aliases.
    count=0
    while IFS=' ' read -r dev point fs options rest; do
        resolved=
        case "$dev" in /dev/block/*) resolved=$(/system/bin/readlink -f "$dev") ;; esac
        if [ "$point" = /cache ]; then
            [ "$resolved" = "$NODE" ] && [ "$fs" = ext4 ] || return 1
            case ",$options," in *,rw,*) ;; *) return 1 ;; esac
            count=$((count + 1))
        elif [ "$resolved" = "$NODE" ]; then
            return 1
        fi
    done < /proc/mounts
    [ "$count" = 1 ]
}

copy_new() {
    src=$1; name=$2
    [ -f "$src" ] && [ ! -L "$src" ] || return 0
    [ ! -e "$DEST/$name" ] && [ ! -L "$DEST/$name" ] || return 0
    /system/bin/head -c 262144 "$src" > "$DEST/$name" || return 1
    source_hash=$(/system/bin/sha256sum "$src") || return 1
    dest_hash=$(/system/bin/sha256sum "$DEST/$name") || return 1
    source_hash=${source_hash%% *}; dest_hash=${dest_hash%% *}
    [ "$source_hash" = "$dest_hash" ] || return 1
    printf '%s  %s\n' "$dest_hash" "$name" >> "$DEST/SHA256SUMS"
}

worker() {
    case "$EXPECTED_CACHE_UUID" in ????????-????-????-????-????????????) ;; *) exit 10 ;; esac
    case "$EXPECTED_CACHE_UUID" in *[!0-9a-f-]*) exit 10 ;; esac
    [ "$(/system/bin/cat /sys/class/block/mmcblk0p36/start)" = 15204376 ] || exit 11
    [ "$(/system/bin/cat /sys/class/block/mmcblk0p36/size)" = 3356672 ] || exit 12
    [ "$(/system/bin/cat /sys/class/block/mmcblk0p36/dev)" = 259:4 ] || exit 13
    identity=$(/system/bin/timeout -k 1 3 /system/bin/toybox blkid "$NODE") || exit 14
    case "$identity" in *'TYPE="ext4"'*) ;; *) exit 15 ;; esac
    case "$identity" in *"UUID=\"$EXPECTED_CACHE_UUID\""*) ;; *) exit 16 ;; esac
    cache_mount_state || exit 17
    # No reuse across boots: preserve an earlier test's evidence on disk.
    /system/bin/mkdir "$DEST" || exit 18
    set -C
    printf 'Test22 boot snapshots; no mount or format performed by logger.\n' > "$DEST/ABOUT.txt"
    printf '' > "$DEST/SHA256SUMS"
    n=0
    while [ "$n" -lt 135 ]; do
        n=$((n + 1))
        for snap in 01 02 03 04; do
            for kind in identity dmesg logcat usb processes stacks graphics; do
                copy_new "$RAM/$snap-$kind.ready" "$snap-$kind.txt" || exit 19
            done
        done
        if [ -f "$RAM/collector.ready" ]; then
            copy_new "$RAM/collector.ready" collector.txt || exit 20
            printf 'Available fixed snapshots copied and hash-checked.\n' > "$DEST/COMPLETE.txt"
            /system/bin/sync
            exit 0
        fi
        /system/bin/sleep 1
    done
    printf 'Collector deadline reached; available files retained.\n' > "$DEST/INCOMPLETE.txt"
    /system/bin/sync
    exit 21
}

case "$1" in
    --worker) worker ;;
    '')
        /system/bin/setsid /system/bin/timeout -k 2 145 /system/bin/sh "$SELF" --worker &
        pid=$!
        /system/bin/sleep 150
        /system/bin/kill -TERM -- "-$pid" 2>/dev/null
        /system/bin/kill -KILL -- "-$pid" 2>/dev/null
        exit 0
        ;;
    *) exit 2 ;;
esac

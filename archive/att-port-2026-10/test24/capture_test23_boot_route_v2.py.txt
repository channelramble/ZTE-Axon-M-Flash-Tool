from pathlib import Path
import time,json,os
from capture_test22_adb import ADB,PORT,ROOT,Capture,bounded_process,identity_command,verify_identity,write_text_capture,save_state
os.umask(0o077)
out=ROOT/'test23-boot-route-v2';out.mkdir(exist_ok=False)
state={'files':{},'read_only':True,'complete':False};save_state(out,state)
raw,rc,status=bounded_process([str(ADB),'-P',PORT,'devices'],10,65536)
rows=[x.split() for x in raw.decode().splitlines()[1:] if len(x.split())>=2]
assert len(rows)==1 and rows[0][1]=='device'
c=Capture(rows[0][0],time.monotonic()+180)
b,info=c.remote(identity_command(),4096,10);assert verify_identity(b,info)
commands=[
('pstore-previous.txt','cat /sys/fs/pstore/console-ramoops-0',524288,10),
('gapps-preflight.txt','getprop ro.build.version.release; getprop ro.build.version.sdk; getprop ro.product.cpu.abi; getprop ro.lineage.version; settings get global device_provisioned; settings get secure user_setup_complete; df -k /system /system_ext /product /vendor /data; pm list packages -s | grep -E "google|gms|vending|setupwizard"; true',32768,10),
('data-recovery-index.txt','ls -la /data/misc/recovery; ls -l /data/misc/recovery/proc; getprop ro.bootmode; getprop sys.boot_completed',16384,10)
]
for name,command,cap,seconds in commands:write_text_capture(c,out,state,name,command,cap,seconds)
state['complete']=True;save_state(out,state);print('Read-only boot routing capture complete',flush=True)

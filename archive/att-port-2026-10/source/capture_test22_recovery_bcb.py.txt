#!/usr/bin/env python3
"""Read-only Test22 BCB recovery capture; local validation unless --execute.

Requires exact native Test19 recovery running in recovery mode, unchanged boot,
and pinned partition geometry. No mount, writes, service control or reboot.
Device identity values and userdata contents are never printed or persisted.
"""
from pathlib import Path
import argparse
import base64
import hashlib
import json
import os
import re
import secrets
import selectors
import shlex
import subprocess
import time

ROOT = Path(__file__).resolve().parent
ADB = Path.home() / '.local/share/axon-m-edl/platform-tools-29.0.6/adb'
PORT = '5038'
KERNEL = '4.4.302-cip112-fujisan'
WHOLE_SECONDS = 240
RECOVERY_SHA = '492b14d1ef9d3f269716465d4d492689e6a41df018f86e4efac139bf34bee220'
BOOT_SHA = '4434306a5c58d4de5ee018a59a04ae3f551d7073efae1fe7a07bd07e468a17d6'
MISC_SHA = '07e3021342d8a12056e9faf36ade8832e1d24e8daaa6930516da3ed88412183e'
MISC_SIZE = 1048576
GEOMETRY = {'boot': (18, 1050624, 131072, '179:18'),
            'misc': (38, 18626584, 2048, '259:6'),
            'userdata': (48, 19083264, 103059423, '259:16')}

def sha(data):
    return hashlib.sha256(data).hexdigest()


def parse_envelope(raw, begin, end, cap):
    """Ignore adb/propClient preamble; accept only this request's frame."""
    opening = b'\n' + begin.encode() + b'\n'
    at = raw.find(opening)
    if at < 0:
        return b'', None, False
    content = raw[at + len(opening):]
    marker = b'\n' + end.encode() + b':'
    finish = content.rfind(marker)
    if finish < 0:
        return content[:cap], None, False
    suffix = content[finish + len(marker):]
    match = re.match(rb'([0-9]{1,3})\r?\n', suffix)
    if not match or finish > cap:
        return content[:cap], None, False
    return content[:finish], int(match[1]), True


def bounded_process(command, timeout, cap):
    process = subprocess.Popen(command, stdout=subprocess.PIPE, stderr=subprocess.STDOUT)
    stream = process.stdout
    os.set_blocking(stream.fileno(), False)
    selector = selectors.DefaultSelector()
    selector.register(stream, selectors.EVENT_READ)
    data = bytearray()
    stop = time.monotonic() + timeout
    status = 'ok'
    try:
        while selector.get_map():
            left = stop - time.monotonic()
            if left <= 0:
                status = 'host_timeout'
                break
            for key, _ in selector.select(min(left, 0.2)):
                chunk = os.read(key.fd, min(65536, cap + 1 - len(data)))
                if not chunk:
                    selector.unregister(key.fileobj)
                    continue
                data.extend(chunk)
                if len(data) > cap:
                    status = 'byte_limit'
                    break
            if status != 'ok':
                break
        if status != 'ok':
            process.kill()
        try:
            result = process.wait(timeout=max(0.1, min(2, stop - time.monotonic())))
        except subprocess.TimeoutExpired:
            process.kill()
            result = process.wait(timeout=2)
            status = 'host_timeout'
    finally:
        selector.close()
        stream.close()
        if process.poll() is None:
            process.kill()
            process.wait(timeout=2)
    return bytes(data[:cap]), result, status


class Capture:
    def __init__(self, serial, deadline):
        self.serial = serial  # Bound internally; never printed or recorded.
        self.deadline = deadline

    def remote(self, command, cap=262144, seconds=8):
        left = self.deadline - time.monotonic()
        if left < 2:
            return b'', {'status': 'transaction_deadline', 'remote_rc': None}
        remote_seconds = max(1, min(seconds, int(left - 1)))
        nonce = secrets.token_hex(12)
        begin, end = 'AXON22_BEGIN_' + nonce, 'AXON22_END_' + nonce
        wrapper = (
            'set -o pipefail; '
            f"printf '\\n%s\\n' {shlex.quote(begin)}; "
            f'/system/bin/timeout -k 1 {remote_seconds} /system/bin/sh -c '
            f'{shlex.quote(command)} 2>&1 | /system/bin/head -c {cap}; '
            'rc=$?; '
            f"printf '\\n%s:%s\\n' {shlex.quote(end)} \"$rc\""
        )
        raw, adb_rc, status = bounded_process(
            [str(ADB), '-P', PORT, '-s', self.serial, 'exec-out',
             '/system/bin/sh -c ' + shlex.quote(wrapper)],
            min(left, remote_seconds + 4), cap + 8192)
        body, remote_rc, framed = parse_envelope(raw, begin, end, cap)
        if status == 'ok' and not framed:
            status = 'framing_error'
        elif status == 'ok' and (adb_rc != 0 or remote_rc != 0):
            status = 'remote_error'
        return body, {'status': status, 'remote_rc': remote_rc, 'adb_rc': adb_rc,
                      'framed': framed, 'bytes': len(body)}


def save_state(out, state):
    temporary = out / '.capture.json.tmp'
    with temporary.open('w') as stream:
        json.dump(state, stream, indent=2)
        stream.write('\n')
        stream.flush()
        os.fsync(stream.fileno())
    temporary.replace(out / 'capture.json')


def require(condition, message):
    if not condition:
        raise ValueError(message)


def file_sha(path):
    h = hashlib.sha256()
    with path.open('rb') as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b''):
            h.update(block)
    return h.hexdigest()


def local_baseline(before):
    for name, size, digest in (('boot', 67108864, BOOT_SHA), ('misc', MISC_SIZE, MISC_SHA)):
        p = before / (name + '.bin')
        require(p.is_file() and not p.is_symlink() and p.stat().st_size == size,
                'Missing exact local ' + name + ' baseline')
        require(file_sha(p) == digest, 'Local baseline hash differs: ' + name)
    data = (before / 'misc.bin').read_bytes()
    require(not any(data[:2048]), 'Baseline BCB is not empty')
    return data


def recovery_identity_command():
    # Full cmdline, process ID and all unrelated properties remain on device.
    return (
        'set -e; '
        "printf 'kernel=%s\n' \"$(/system/bin/uname -r)\"; "
        "printf 'uid=%s\n' \"$(/system/bin/id -u)\"; "
        "printf 'recovery=%s\n' \"$(/system/bin/getprop init.svc.recovery)\"; "
        "printf 'twrp=%s\n' \"$(/system/bin/getprop ro.twrp.version)\"; "
        'p=$(/system/bin/pidof recovery); '
        'case "$p" in ""|*[!0-9]*) exit 31 ;; esac; '
        '[ "$(/system/bin/readlink /proc/$p/exe)" = /system/bin/recovery ] || exit 32; '
        'h=$(/system/bin/sha256sum /proc/$p/exe); '
        "printf 'running_recovery_sha256=%s\n' \"${h%% *}\"; "
        'h=$(/system/bin/sha256sum /system/bin/recovery); '
        "printf 'recovery_sha256=%s\n' \"${h%% *}\"; "
        'case " $(/system/bin/cat /proc/cmdline) " in '
        '*" androidboot.mode=recovery "*) printf "cmdline_recovery=1\n" ;; '
        '*) exit 33 ;; esac; '
        '[ "$(/system/bin/pidof recovery)" = "$p" ] || exit 34'
    )


def verify_recovery_identity(body, info):
    if info.get('status') != 'ok':
        return False
    try:
        pairs = [line.split('=', 1) for line in body.decode('ascii').splitlines()]
        if any(len(pair) != 2 for pair in pairs):
            return False
        values = dict(pairs)
        return len(pairs) == len(values) and values == {
            'kernel': KERNEL, 'uid': '0', 'recovery': 'running', 'twrp': '',
            'running_recovery_sha256': RECOVERY_SHA,
            'recovery_sha256': RECOVERY_SHA, 'cmdline_recovery': '1'}
    except (UnicodeError, ValueError):
        return False


def geometry_command():
    commands = ['set -e']
    for name, (index, start, sectors, dev) in GEOMETRY.items():
        node = '/dev/block/mmcblk0p' + str(index)
        sys = '/sys/class/block/mmcblk0p' + str(index)
        commands += [f'[ -b {node} ]',
                     f'[ "$(/system/bin/readlink -f /dev/block/bootdevice/by-name/{name})" = {node} ]',
                     f'[ "$(/system/bin/cat {sys}/start)" = {start} ]',
                     f'[ "$(/system/bin/cat {sys}/size)" = {sectors} ]',
                     f'[ "$(/system/bin/cat {sys}/dev)" = {dev} ]',
                     f'[ "$(/system/bin/blockdev --getsize64 {node})" = {sectors * 512} ]']
    commands.append("printf 'geometry=verified\n'")
    return '; '.join(commands)


def boot_hash_command():
    return ('set -e; h=$(/system/bin/sha256sum /dev/block/mmcblk0p18); '
            "printf 'boot_sha256=%s\n' \"${h%% *}\"")


def misc_command():
    return (
        'set -e; p=/dev/block/mmcblk0p38; '
        '[ "$(/system/bin/blockdev --getsize64 "$p")" = 1048576 ]; '
        'h=$(/system/bin/sha256sum "$p"); '
        "printf 'SIZE 1048576\nSHA %s\n' \"${h%% *}\"; "
        '/system/bin/base64 "$p"'
    )


def decode_misc(body):
    lines = body.split(b'\n', 2)
    require(len(lines) == 3 and lines[0] == b'SIZE 1048576', 'Bad misc size envelope')
    require(re.fullmatch(rb'SHA [0-9a-f]{64}', lines[1]), 'Bad misc hash envelope')
    encoded = lines[2].replace(b'\n', b'').replace(b'\r', b'')
    data = base64.b64decode(encoded, validate=True)
    require(len(data) == MISC_SIZE and sha(data).encode() == lines[1][4:],
            'Truncated, changed or corrupted misc capture')
    return data


def check_misc(data, baseline):
    require(len(data) == MISC_SIZE and len(baseline) == MISC_SIZE, 'Wrong misc size')
    return {'full_misc_sha256': sha(data), 'bcb_first_2048_zero': not any(data[:2048]),
            'misc_above_2048_preserved': data[2048:] == baseline[2048:],
            'full_misc_restored_to_baseline': data == baseline and sha(data) == MISC_SHA}


def userdata_command():
    # Read only public geometry/FS type and the first 8KiB digest; never UUIDs,
    # file names, directory contents, or the raw superblocks themselves.
    return (
        'set -e; p=/dev/block/mmcblk0p48; '
        "printf 'bytes=%s\n' \"$(/system/bin/blockdev --getsize64 \"$p\")\"; "
        'a=$(/system/bin/od -An -tx1 -j 1024 -N 4 "$p"); '
        'b=$(/system/bin/od -An -tx1 -j 5120 -N 4 "$p"); '
        'case "$a|$b" in *"10 20 f5 f2"*"|"*"10 20 f5 f2"*) '
        'printf "both_f2fs_magic=1\n" ;; *) printf "both_f2fs_magic=0\n" ;; esac; '
        'h=$(/system/bin/head -c 8192 "$p" | /system/bin/sha256sum); '
        "printf 'superblock_prefix_sha256=%s\n' \"${h%% *}\"; "
        "/system/bin/awk 'BEGIN { n=0 } $2 == \"/data\" { n++; "
        "printf \"data_mount_type=%s\\ndata_mount_options=%s\\n\", $3, $4 } "
        "END { printf \"data_mount_count=%d\\n\", n }' /proc/mounts"
    )


def execute(before, out):
    baseline = local_baseline(before)
    out.mkdir(mode=0o700, parents=True, exist_ok=False)
    state = {'complete': False, 'read_only': True, 'no_device_writes_mounts_or_reset': True,
             'whole_timeout_seconds': WHOLE_SECONDS, 'files': {},
             'baseline_misc_sha256': MISC_SHA, 'expected_boot_sha256': BOOT_SHA}
    save_state(out, state)
    deadline = time.monotonic() + WHOLE_SECONDS
    try:
        raw, rc, status = bounded_process([str(ADB), '-P', PORT, 'devices'], 10, 65536)
        rows = [line.split() for line in raw.decode('utf-8', 'replace').splitlines()
                if '\t' in line and len(line.split()) >= 2]
        require(status == 'ok' and rc == 0 and len(rows) == 1
                and rows[0][1] in ('device', 'recovery'), 'Exactly one usable adb target required')
        capture = Capture(rows[0][0], deadline)
        for phase in ('before', 'after'):
            body, info = capture.remote(recovery_identity_command(), 4096, 15)
            require(verify_recovery_identity(body, info), 'Exact running recovery identity guard failed')
            state['recovery_identity_' + phase + '_verified'] = True
            body, info = capture.remote(geometry_command(), 4096, 15)
            require(info['status'] == 'ok' and body == b'geometry=verified\n', 'Partition geometry guard failed')
            body, info = capture.remote(boot_hash_command(), 4096, 60)
            require(info['status'] == 'ok' and body == ('boot_sha256=' + BOOT_SHA + '\n').encode(),
                    'Full boot hash is not unchanged Test20')
            state['full_boot_' + phase + '_verified'] = True
            save_state(out, state)
            if phase == 'after':
                break
            body, info = capture.remote(misc_command(), 1500000, 30)
            require(info['status'] == 'ok', 'Misc capture failed or was truncated')
            data = decode_misc(body)
            with (out / 'misc.bin').open('xb') as stream:
                stream.write(data); stream.flush(); os.fsync(stream.fileno())
            state['files']['misc.bin'] = {'sha256': sha(data), 'bytes': len(data), 'status': 'ok'}
            state.update(check_misc(data, baseline))
            save_state(out, state)
            body, info = capture.remote(userdata_command(), 8192, 15)
            # Fixed diagnostics contain no raw device identifiers or user files.
            if info['status'] == 'ok':
                with (out / 'userdata-info.txt').open('xb') as stream:
                    stream.write(body)
                state['files']['userdata-info.txt'] = {'sha256': sha(body), 'bytes': len(body), 'status': 'ok'}
            else:
                state['userdata_info_status'] = info['status']
            save_state(out, state)
        require(state['bcb_first_2048_zero'] and state['misc_above_2048_preserved']
                and state['full_misc_restored_to_baseline'],
                'BCB not cleared or unrelated misc bytes changed; no device action taken')
        state['complete'] = True
        save_state(out, state)
        print(json.dumps({'complete': True, 'recovery_identity_verified': True,
                          'boot_unchanged': True, 'bcb_cleared': True,
                          'higher_misc_preserved': True, 'no_device_writes_or_reset': True}), flush=True)
        return 0
    except Exception as exc:
        # Fixed local errors only: never print the device response or serial.
        state['error'] = type(exc).__name__ + ': ' + str(exc)
        save_state(out, state)
        print(json.dumps({'complete': False, 'error': state['error'], 'no_device_writes_or_reset': True}), flush=True)
        return 2


def main():
    os.umask(0o077)
    p = argparse.ArgumentParser(description=__doc__)
    p.add_argument('--before', type=Path, default=ROOT / 'test22-edl-before')
    p.add_argument('--output-dir', type=Path, default=ROOT / 'test22-recovery-bcb-capture')
    p.add_argument('--execute', action='store_true')
    args = p.parse_args()
    local_baseline(args.before)
    if not args.execute:
        print(json.dumps({'files_only_validated': True, 'phone_operations': False,
                          'expected_recovery_sha256': RECOVERY_SHA,
                          'expected_boot_sha256': BOOT_SHA, 'expected_misc_sha256': MISC_SHA}))
        return 0
    return execute(args.before, args.output_dir)


if __name__ == '__main__':
    raise SystemExit(main())

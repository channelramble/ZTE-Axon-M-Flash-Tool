#!/usr/bin/env python3
"""Local-only same-size firmware data patch in an APFS clone of LOS ext4.
Pin source bytes, verify file blocks against debugfs dumps, retain every inode,
xattr and other image byte. No filesystem mutation via debugfs or phone access.
"""
from pathlib import Path
import hashlib,json,os,re,subprocess
ROOT=Path(__file__).resolve().parent
DEBUGFS='/opt/homebrew/opt/e2fsprogs/sbin/debugfs'
SRC_SHA='90aea590dd4921df4bd78ebcdcd6c10d93eebe27e8d70d501d8a59e7f495899d'
sha=lambda b:hashlib.sha256(b).hexdigest()
def file_sha(p):
 h=hashlib.sha256()
 with p.open('rb') as f:
  for b in iter(lambda:f.read(1048576),b''):h.update(b)
 return h.hexdigest()
def debug(image,cmd):
 p=subprocess.run([DEBUGFS,'-R',cmd,str(image)],capture_output=True,text=True,check=True)
 assert not any(x in p.stderr.lower() for x in ('not found','error','invalid')),p.stderr
 return p.stdout

def main():
 os.umask(0o077)
 src=ROOT/'los23-system-original/system.img';assert file_sha(src)==SRC_SHA
 out=ROOT/'los23-system-att-gpu';assert not out.exists();out.mkdir(mode=0o700)
 expected=json.loads((ROOT/'hardware-tests/test06-reader-control/live-baseline/system-gpu-firmware.manifest.json').read_text())
 patches=[];files=[];covered=set()
 with src.open('rb') as f:
  for item in expected:
   name=Path(item['name']).name;assert re.fullmatch(r'a530_zap\.(?:mdt|elf|b00|b01|b02)',name)
   att=(ROOT/'los23-gpu-analysis'/('att-'+name)).read_bytes();old=(ROOT/'los23-gpu-analysis'/('los-'+name)).read_bytes()
   assert len(att)==len(old)==item['bytes'] and sha(att)==item['sha256']
   path='/vendor/firmware/'+name;stat=debug(src,'stat '+path)
   assert 'Type: regular' in stat and 'Flags: 0x80000' in stat and 'Links: 1 ' in stat
   assert int(re.search(r'\bSize: (\d+)',stat).group(1))==len(att)
   blocks=[]
   for logical in range((len(att)+4095)//4096):
    value=debug(src,f'bmap {path} {logical}').strip();assert value.isdigit()
    physical=int(value);assert physical>0 and physical not in covered;covered.add(physical);blocks.append(physical)
    start=logical*4096;n=min(4096,len(att)-start);off=physical*4096
    f.seek(off);assert f.read(n)==old[start:start+n]
    if att[start:start+n]!=old[start:start+n]:patches.append((off,old[start:start+n],att[start:start+n]))
   files.append({'name':name,'bytes':len(att),'source_sha256':sha(old),'att_sha256':sha(att),'data_blocks':blocks,'stat_before':stat})
 dst=out/'system.img';subprocess.run(['/bin/cp','-c',str(src),str(dst)],check=True)
 assert file_sha(dst)==SRC_SHA and dst.stat().st_size==src.stat().st_size
 with dst.open('r+b') as f:
  for off,old,new in patches:
   f.seek(off);assert f.read(len(old))==old;f.seek(off);f.write(new)
  f.flush();os.fsync(f.fileno())
 # Compare complete streams against original with only explicit patch intervals.
 patchmap={off:(old,new) for off,old,new in patches};ordered=sorted(patches);allowed_changed=0;cursor=0
 with src.open('rb') as a,dst.open('rb') as b:
  for off,old,new in ordered+[(src.stat().st_size,b'',b'')]:
   left=off-cursor;assert left>=0
   while left:
    n=min(left,1048576);assert a.read(n)==b.read(n);left-=n
   assert a.read(len(old))==old and b.read(len(new))==new
   allowed_changed+=sum(x!=y for x,y in zip(old,new));cursor=off+len(old)
 for item in files:
  name=item['name'];path='/vendor/firmware/'+name
  assert debug(dst,'stat '+path)==item.pop('stat_before')
  dump=out/name;debug(dst,'dump '+path+' '+str(dump));assert sha(dump.read_bytes())==item['att_sha256']
 check=subprocess.run(['/opt/homebrew/opt/e2fsprogs/sbin/e2fsck','-fn',str(dst)],capture_output=True,text=True)
 (out/'e2fsck-read-only.txt').write_text(check.stdout+check.stderr);assert check.returncode==0
 assert file_sha(src)==SRC_SHA
 result={'source_sha256':SRC_SHA,'image_sha256':file_sha(dst),'image_bytes':dst.stat().st_size,'files':files,'changed_bytes':allowed_changed,'other_image_bytes_exact':True,'all_inode_metadata_xattrs_exact':True,'all_file_readbacks_match_verified_att_captures':True,'read_only_fsck_passed':True,'phone_operations':False,'boot_recovery_coordination_still_required':True}
 (out/'manifest.json').write_text(json.dumps(result,indent=2)+'\n');print(json.dumps({k:v for k,v in result.items() if k!='files'},indent=2))
if __name__=='__main__':main()

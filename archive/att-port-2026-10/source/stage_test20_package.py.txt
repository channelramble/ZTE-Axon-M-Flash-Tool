#!/usr/bin/env python3
"""Stage package files on MacBook only. No adb or phone access."""
from pathlib import Path
import hashlib,json,shlex,subprocess,time,zlib
ROOT=Path(__file__).resolve().parent
CONFIG='/EXAMPLE/LOCAL_HOME/.crosstalk/ssh/config'
REMOTE='/EXAMPLE/REMOTE_HOME/.local/share/axon-m-edl/sessions/SESSION_ID/test20-package-v1'
PY='/EXAMPLE/REMOTE_HOME/.local/share/axon-m-edl/.venv/bin/python'
EXPECTED='7c32f735219768f31444400c67cf996b5f80de4c17d0d1c1ed65a7d0acf0883a'
SIZE=6979321856
def sshcode(code):
 return ['ssh','-F',CONFIG,'-o','BatchMode=yes','-o','ConnectTimeout=10','DEVICE_HOST',
         '$SHELL -lc '+shlex.quote(PY+' -c '+shlex.quote(code))]
def main():
 source=ROOT/'test20-system-v1/system.img'
 manifest=json.loads((source.parent/'manifest.json').read_text())
 assert manifest['passed'] and manifest['image_sha256']==EXPECTED and source.stat().st_size==SIZE
 setup=f"from pathlib import Path; import os,shutil; os.umask(0o077); p=Path({REMOTE!r}); assert shutil.disk_usage(p.parent).free>{SIZE+2*1024**3}; p.mkdir(mode=0o700,exist_ok=False)"
 subprocess.run(sshcode(setup),check=True,timeout=30)
 items=[(ROOT/'test20-boot-v1/boot.img','boot.img'),
        (ROOT/'test20-boot-v1/recovery-control.img','recovery-control.img'),
        (ROOT/'test20-boot-v1/boot.validation.json','boot.validation.json'),
        (ROOT/'test20-boot-v1/recovery-control.validation.json','recovery-control.validation.json'),
        (source.parent/'manifest.json','system.manifest.json')]
 for local,name in items:
  subprocess.run(['scp','-q','-F',CONFIG,str(local),'DEVICE_HOST:'+REMOTE+'/'+name],check=True,timeout=120)
 receiver=f'''from pathlib import Path
import gzip,hashlib,json,os,sys
os.umask(0o077)
p=Path({REMOTE!r});h=hashlib.sha256();count=0;part=p/'system.img.part'
with part.open('xb') as out,gzip.GzipFile(fileobj=sys.stdin.buffer,mode='rb') as inp:
 for b in iter(lambda:inp.read(1048576),b''):
  count+=len(b);assert count<={SIZE};h.update(b)
  if b==b'\\0'*len(b):out.seek(len(b),1)
  else:out.write(b)
 assert count=={SIZE} and h.hexdigest()=={EXPECTED!r}
 out.truncate(count);out.flush();os.fsync(out.fileno())
verify=hashlib.sha256()
with part.open('rb') as f:
 for b in iter(lambda:f.read(4194304),b''):verify.update(b)
assert verify.hexdigest()=={EXPECTED!r}
part.rename(p/'system.img')
for name in ('boot','recovery-control'):
 m=json.loads((p/(name+'.validation.json')).read_text())
 b=(p/(name+'.img')).read_bytes()
 assert len(b)==m['bytes'] and hashlib.sha256(b).hexdigest()==m['image_sha256']
r={{'complete':True,'system_sha256':verify.hexdigest(),'bytes':count,'saved_copy_readback_verified':True,'boot_and_recovery_hashes_verified':True,'phone_operations':False}}
(p/'staging-verification.json').write_text(json.dumps(r,indent=2)+'\\n');print(json.dumps(r),flush=True)
'''
 p=subprocess.Popen(sshcode(receiver),stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE)
 h=hashlib.sha256();count=0;progress=1024**3;start=time.monotonic();compress=zlib.compressobj(1,zlib.DEFLATED,31)
 try:
  with source.open('rb') as f:
   for b in iter(lambda:f.read(1048576),b''):
    assert time.monotonic()-start<1200
    h.update(b);count+=len(b);data=compress.compress(b)
    if data:p.stdin.write(data)
    if count>=progress:print(json.dumps({'raw_bytes_staged':count,'total':SIZE}),flush=True);progress+=1024**3
  assert count==SIZE and h.hexdigest()==EXPECTED
  p.stdin.write(compress.flush());p.stdin.close();p.stdin=None
  out,err=p.communicate(timeout=180)
  assert p.returncode==0,err.decode();print(out.decode(),flush=True)
 finally:
  if p.poll() is None:p.kill();p.wait()
 receipt=ROOT/'test20-staging-verification.json'
 assert not receipt.exists()
 subprocess.run(['scp','-q','-F',CONFIG,'DEVICE_HOST:'+REMOTE+'/staging-verification.json',str(receipt)],check=True,timeout=30)
 result=json.loads(receipt.read_text());assert result['complete'] and result['system_sha256']==EXPECTED
if __name__=='__main__':main()

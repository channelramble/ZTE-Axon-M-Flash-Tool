from pathlib import Path
import time,json,os
from capture_test22_adb import ADB,PORT,ROOT,Capture,bounded_process,identity_command,verify_identity,write_text_capture,save_state
os.umask(0o077)
out=ROOT/'test23-usb-before-v1';out.mkdir(exist_ok=False)
state={'files':{},'read_only':True,'complete':False};save_state(out,state)
raw,rc,status=bounded_process([str(ADB),'-P',PORT,'devices'],10,65536)
rows=[x.split() for x in raw.decode().splitlines()[1:] if len(x.split())>=2]
assert len(rows)==1 and rows[0][1]=='device'
c=Capture(rows[0][0],time.monotonic()+180)
b,info=c.remote(identity_command(),4096,10);assert verify_identity(b,info)
commands=[('usb-service-before.txt', 'getprop init.svc.vendor.usb-hal-1-0; sha256sum /vendor/bin/hw/android.hardware.usb@1.0-service /vendor/etc/init/android.hardware.usb@1.0-service.rc; cat /vendor/etc/init/android.hardware.usb@1.0-service.rc; getprop sys.usb.config; getprop sys.usb.state; getprop persist.sys.usb.config; cat /config/usb_gadget/g1/UDC; ls -l /vendor/bin/hw/android.hardware.usb*', 16384, 8), ('usb-declarations.txt', 'grep -n -A 12 -B 3 android.hardware.usb /vendor/etc/vintf/manifest.xml /vendor/etc/vintf/manifest/*.xml', 32768, 8)]
for name,command,cap,seconds in commands:write_text_capture(c,out,state,name,command,cap,seconds)
state['complete']=True;save_state(out,state);print('Black-screen capture complete',flush=True)

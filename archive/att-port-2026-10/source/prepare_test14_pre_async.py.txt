#!/usr/bin/env python3
"""Local-only milestone after kernel_init_freeable, before original async wait."""
import gzip,hashlib,json,os,struct,subprocess,sys,tempfile
from pathlib import Path
ROOT=Path(__file__).resolve().parent
sys.path.insert(0,str(ROOT.parent/'ZTE-Axon-M-Flash-Tool/scripts'))
from fujisan_image import BootImage,gzip_member,report,sign_image
from prepare_test12_preflush import bl as branch
SHA=lambda b:hashlib.sha256(b).hexdigest()
BASE=0xffffff8008080000
CAVE=0xffffff80097867e8
EXTERNAL={'green_sync':0xffffff80097866d8,'async_synchronize_full':0xffffff80080cfbd8}

def link_object(data):
    if data[:7]!=b'\x7fELF\x02\x01\x01' or struct.unpack_from('<HH',data,16)!=(1,183):
        raise ValueError('Expected little-endian arm64 relocatable ELF')
    shoff=struct.unpack_from('<Q',data,40)[0]
    entsize,count,strindex=struct.unpack_from('<HHH',data,58)
    assert entsize==64
    sections=[struct.unpack_from('<IIQQQQIIQQ',data,shoff+i*64) for i in range(count)]
    content=lambda s:data[s[4]:s[4]+s[5]]
    strings=content(sections[strindex])
    cstr=lambda table,pos:table[pos:table.index(b'\0',pos)].decode()
    named={cstr(strings,s[0]):(i,s) for i,s in enumerate(sections)}
    textindex,textsection=named['.text'];code=bytearray(content(textsection))
    syms=content(named['.symtab'][1]);names=content(sections[named['.symtab'][1][6]])
    symbols=[]
    for off in range(0,len(syms),24):
        name,info,other,shndx,value,size=struct.unpack_from('<IBBHQQ',syms,off)
        symbols.append((cstr(names,name),shndx,value))
    addresses={name:CAVE+value for name,idx,value in symbols if idx==textindex and not name.startswith('$')}
    assert addresses['pre_async_marker']==CAVE
    assert set(addresses)=={'pre_async_marker'}
    calls=[]
    for name,(i,s) in named.items():
        if s[1] not in (4,9):continue
        assert name=='.rela.text' and s[1]==4 and s[7]==textindex and s[9]==24
        rel=content(s)
        for off in range(0,len(rel),24):
            offset,info,addend=struct.unpack_from('<QQq',rel,off)
            assert info&0xffffffff==283 and addend==0
            symbol,idx,value=symbols[info>>32]
            target=addresses[symbol] if idx==textindex else EXTERNAL[symbol]
            assert code[offset:offset+4]==bytes.fromhex('00000094')
            code[offset:offset+4]=branch(CAVE+offset,target)
            calls.append(dict(offset=hex(offset),symbol=symbol,target=hex(target)))
    assert len(calls)==2
    return bytes(code),addresses,calls

def main():
    os.umask(0o077)
    source=ROOT/'prepared/test-13-deferred-observability-boot-adb-only.img'
    target=ROOT/'prepared/test-14-pre-async-green-boot-adb-only.img'
    paths=[target,target.with_suffix('.validation.json'),target.with_suffix('.img.json')]
    assert not any(p.exists() for p in paths)
    data=source.read_bytes()
    assert SHA(data)=='5181b02e77574c97986ca989d04e7e61336c0d8ded9651c3337e206bb8012129'
    boot=BootImage(data);_,raw,tail=gzip_member(boot.kernel)
    assert SHA(raw)=='39fcd246f403913f9840c2af80fefbea35f55253fe1cbde34fa77ccc4669fde0'
    with tempfile.TemporaryDirectory() as d:
        obj=Path(d)/'helper.o'
        subprocess.run(['/usr/bin/clang','--target=aarch64-none-elf','-c',str(ROOT/'test14_pre_async_marker.S'),'-o',str(obj)],check=True)
        code,addresses,calls=link_object(obj.read_bytes())
    assert len(code)==28 and CAVE+len(code)<0xffffff800978684c
    patched=bytearray(raw);patched[CAVE-BASE:CAVE-BASE+len(code)]=code
    patches=[(0xffffff80097867c8,bytes.fromhex('e01f8052'),bytes.fromhex('e0031f2a')),
             (0xffffff8009044538,branch(0xffffff8009044538,EXTERNAL['async_synchronize_full']),branch(0xffffff8009044538,CAVE))]
    for pc,old,new in patches:
        assert raw[pc-BASE:pc-BASE+4]==old
        patched[pc-BASE:pc-BASE+4]=new
    assert patched[0x174be28:0x174d3b8]==raw[0x174be28:0x174d3b8]
    assert patched[0xfc453c:0xfc4570]==raw[0xfc453c:0xfc4570]
    assert patched[0x5febfc:0x5fedac]==raw[0x5febfc:0x5fedac]
    intervals=[(CAVE-BASE,CAVE-BASE+len(code))]+[(pc-BASE,pc-BASE+4) for pc,_,_ in patches]
    changes=[i for i,(a,b) in enumerate(zip(raw,patched)) if a!=b]
    assert all(any(lo<=i<hi for lo,hi in intervals) for i in changes)
    signed=sign_image(boot.rebuild(kernel=gzip.compress(bytes(patched),compresslevel=9,mtime=0)+tail),
                      '/boot',str(ROOT/'keys/axon-m-local.key'),str(ROOT/'keys/axon-m-local.pem'))
    new=BootImage(signed);_,checked,newtail=gzip_member(new.kernel)
    assert checked==patched and newtail==tail and new.ramdisk==boot.ramdisk and new.cmdline==boot.cmdline
    assert boot.rebuild(kernel=new.kernel)==new.payload
    info=report(signed)
    assert info['signature_verified'] and info['boot_id_valid'] and info['signature_target']=='/boot'
    assert all(t['free_tail_bytes']==8192 for t in info['dtbs'])
    v=dict(source_image=source.name,source_sha256=SHA(data),image=target.name,image_bytes=len(signed),
           image_sha256=SHA(signed),padded_sha256=SHA(signed+b'\0'*(-len(signed)%512)),raw_sha256=SHA(patched),
           changed_raw_bytes=len(changes),changed_raw_offsets=changes,permitted_intervals=[[hex(a),hex(b)] for a,b in intervals],
           helper_address=hex(CAVE),helper_bytes=len(code),helper_sha256=SHA(code),helper_calls=calls,
           patches=[dict(address=hex(pc),old=old.hex(),new=new.hex()) for pc,old,new in patches],
           all_690_pointers_identical_to_test13=True,original_async_wait_runs_once=True,
           deferred_bypass_init_memory_and_power_retention_preserved=True,
           pre_enable_pulse_unchanged=True,dt_ramdisk_cmdline_unchanged=True,
           signature_verified=True,boot_id_valid=True,hardware_tested=False,
           interpretation='Conditional deferred-flush bypass now leaves green OFF. After kernel_init_freeable returns, the new helper turns green ON and executes the original async_synchronize_full exactly once. A final green after the prior off supports reaching beyond initcalls and namespace check, before async completion/userspace. Red remains conditional on LED availability/lock contention. This retains Test13 temporary bypass and resource retention, not a production fix.')
    for p,b in [(paths[0],signed),(paths[1],(json.dumps(v,indent=2)+'\n').encode()),(paths[2],(json.dumps(info,indent=2)+'\n').encode())]:
        with p.open('xb') as f:f.write(b)
    with (ROOT/'test14-linked-helper.bin').open('xb') as f:f.write(code)
    print(json.dumps({k:x for k,x in v.items() if k!='changed_raw_offsets'},indent=2))

if __name__=='__main__':main()

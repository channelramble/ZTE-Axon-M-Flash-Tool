from pathlib import Path
import time,json,os
from capture_test22_adb import ADB,PORT,ROOT,Capture,bounded_process,identity_command,verify_identity,write_text_capture,save_state
os.umask(0o077)
out=ROOT/'test22-black-screen-v2';out.mkdir(exist_ok=False)
state={'files':{},'read_only':True,'complete':False};save_state(out,state)
raw,rc,status=bounded_process([str(ADB),'-P',PORT,'devices'],10,65536)
rows=[x.split() for x in raw.decode().splitlines()[1:] if len(x.split())>=2]
assert len(rows)==1 and rows[0][1]=='device'
c=Capture(rows[0][0],time.monotonic()+180)
b,info=c.remote(identity_command(),4096,10);assert verify_identity(b,info)
commands=[('debugger-help.txt', 'debuggerd --help', 8192, 5), ('am-processes.txt', 'dumpsys activity processes', 262144, 12), ('broadcasts.txt', 'dumpsys activity broadcasts', 262144, 12), ('bootstate.txt', 'getprop service.bootanim.exit; getprop init.svc.bootanim; getprop ro.crypto.state; getprop ro.crypto.type; getprop sys.boot_completed; getprop sys.boot.reason; getprop sys.boot.reason.last; getprop persist.sys.boot.reason', 8192, 5), ('thread-waits.txt', 'p=$(pidof system_server); case "$p" in ""|*[!0-9]*) exit 2;; esac; for t in /proc/$p/task/*; do printf "%s " "${t##*/}"; cat "$t/comm" "$t/wchan"; done', 131072, 10), ('anr-index.txt', 'ls -lt /data/anr; dumpsys dropbox --file system_server_watchdog; dumpsys dropbox --file system_server_crash', 65536, 10)]
for name,command,cap,seconds in commands:write_text_capture(c,out,state,name,command,cap,seconds)
state['complete']=True;save_state(out,state);print('Black-screen capture complete',flush=True)

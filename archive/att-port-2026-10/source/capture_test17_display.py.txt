#!/usr/bin/env python3
"""Private read-only live capture if the diagnostic reaches adb. Never reboots."""
import hashlib,json,os,shlex,subprocess,time
from pathlib import Path
ROOT=Path(__file__).resolve().parent
ADB=Path.home()/'.local/share/axon-m-edl/platform-tools-29.0.6/adb'
os.umask(0o077)
r=subprocess.run([str(ADB),'-P','5038','devices'],capture_output=True,text=True,timeout=10)
devices=[x.split() for x in r.stdout.splitlines()[1:] if len(x.split())>=2]
connected=[x for x in devices if x[1] in ('device','recovery')]
if len(connected)!=1:
    print(json.dumps({'captured':False,'adb_states':[x[1] for x in devices]}))
    raise SystemExit(0)
serial=connected[0][0]
out=ROOT/'test17-display-state'
out.mkdir(mode=0o700,exist_ok=False)
captures=[
 ('display-status.txt','for f in /sys/kernel/debug/fujisan_display/a_retire_fence /sys/kernel/debug/fujisan_display/b_retire_fence /sys/kernel/debug/fujisan_display/last_error /sys/kernel/debug/fujisan_display/last_submit /sys/kernel/debug/fujisan_display/mode; do echo "$f"; cat "$f"; done',12),
 ('regulator-summary.txt','cat /sys/kernel/debug/regulator/regulator_summary',12),
 ('framebuffer-status.txt','for d in /sys/class/graphics/fb0 /sys/class/graphics/fb1; do for n in name virtual_size bits_per_pixel modes blank; do echo "$d/$n"; cat "$d/$n" 2>/dev/null; done; done',12),
 ('enabled-clocks.txt','cat /sys/kernel/debug/clk/enabled_clocks',12),
 ('dmesg.txt','dmesg',12)]
state={'captured':True,'adb_state':connected[0][1],'files':{}}
for name,command,timeout in captures:
    path=out/name
    with path.open('xb') as f:
        p=subprocess.Popen([str(ADB),'-P','5038','-s',serial,'exec-out','sh','-c',command],stdout=f,stderr=subprocess.STDOUT)
        try:rc=p.wait(timeout=timeout)
        except subprocess.TimeoutExpired:p.kill();p.wait();rc='timeout'
    data=path.read_bytes()
    state['files'][name]={'bytes':len(data),'sha256':hashlib.sha256(data).hexdigest(),'status':rc}
    (out/'capture.json').write_text(json.dumps(state,indent=2)+'\n')
    print(json.dumps({'file':name,'bytes':len(data),'status':rc}),flush=True)

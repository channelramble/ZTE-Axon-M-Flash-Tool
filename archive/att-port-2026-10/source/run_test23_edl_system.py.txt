#!/usr/bin/env python3
"""Exact Test22-to-Test23 system-only EDL delta, no erase/reset/retry.

Default preflight reads host files only. --execute requires the already-loaded
programmer and a fresh, two-Mac-verified full before capture from this EDL
session. This script never loads a programmer or writes any other partition.
"""
from pathlib import Path
import argparse
import base64
import hashlib
import json
import os
import shutil
import signal
import sys
import xml.etree.ElementTree as ET

ROOT = Path(__file__).resolve().parent
LOCAL_SCRIPTS = ROOT.parent / 'ZTE-Axon-M-Flash-Tool/scripts'
if LOCAL_SCRIPTS.is_dir():
    sys.path.insert(0, str(LOCAL_SCRIPTS))
from check_backups import read_gpt
from firehose_read import Reader

START = 1312768
SECTORS = 13631488
SIZE = 6979321856
SOURCE_SHA = '508cd08359c6602c36d1bf0a2711dc064bfae509b5773b55933d68ad1f2570a8'
TARGET_SHA = '33db1d4d4e5e49f5dc1c06cbd6a0e39b07fd9b6d886941e33f165b7876fccc5a'
DELTA_SHA = '8fcfb957f737c58e3510a0d87150121a0805eaa59560ab955b5aa136c9af142b'
OFFSETS = (2215178240,)
BOUNDS = {'gpt': (0, 14), 'boot': (1050624, 131072),
          'recovery': (1181696, 131072), 'misc': (18626584, 2048),
          'system': (START, SECTORS)}
PINS = {'gpt': '7f1e4fd44e69e004189f0256ce6d35addfb1e3dbfe1cd55bfd2047b580b1e891',
        'boot': '4434306a5c58d4de5ee018a59a04ae3f551d7073efae1fe7a07bd07e468a17d6',
        'recovery': 'f5bb5bf5d70ab1e44fd5ac9d888270e219f527d21a67796cf683cef5521ecb35',
        'misc': '07e3021342d8a12056e9faf36ade8832e1d24e8daaa6930516da3ed88412183e',
        'system': SOURCE_SHA}
sha = lambda b: hashlib.sha256(b).hexdigest()


def require(condition, message):
    if not condition:
        raise ValueError(message)


def file_sha(path):
    h = hashlib.sha256()
    with path.open('rb') as f:
        for chunk in iter(lambda: f.read(4 * 1024 * 1024), b''):
            h.update(chunk)
    return h.hexdigest()


def checked_file(path, size, digest):
    require(path.is_file() and not path.is_symlink(), 'Missing/nonregular file: ' + str(path))
    require(path.stat().st_size == size and file_sha(path) == digest,
            'File size/hash mismatch: ' + str(path))


def load(path):
    require(path.is_file() and not path.is_symlink(), 'Missing metadata: ' + str(path))
    return json.loads(path.read_text())


def save(path, state):
    temporary = path.with_name(path.name + '.tmp')
    with temporary.open('w') as f:
        json.dump(state, f, indent=2)
        f.write('\n')
        f.flush()
        os.fsync(f.fileno())
    os.replace(temporary, path)
    fd = os.open(path.parent, os.O_RDONLY)
    try:
        os.fsync(fd)
    finally:
        os.close(fd)


def delta_blocks(data):
    require(sha(data) == DELTA_SHA, 'Delta manifest is not the reviewed exact artifact')
    manifest = json.loads(data)
    require(manifest['source_sha256'] == SOURCE_SHA and manifest['image_sha256'] == TARGET_SHA
            and manifest['bytes'] == SIZE, 'Wrong delta image identity')
    rows = manifest['blocks']
    require(tuple(row['offset'] for row in rows) == OFFSETS, 'Wrong block offsets/count/order')
    result = []
    for row in rows:
        offset = row['offset']
        require(type(offset) is int and offset % 4096 == 0 and 0 <= offset <= SIZE - 4096,
                'Block outside fixed system bounds')
        block = base64.b64decode(row['data_base64'], validate=True)
        require(len(block) == 4096 and sha(block) == row['sha256'], 'Wrong block data')
        require(row['original_sha256'] != row['sha256'], 'Unchanged delta block')
        result.append((offset, row['original_sha256'], block))
    return result


def preflight(package, before):
    blocks = delta_blocks((ROOT / 'test23-system-delta.json').read_bytes())
    checked_file(package / 'system.img', SIZE, TARGET_SHA)
    staged = load(package / 'staging-verification.json')
    require(staged['complete'] and staged['saved_copy_readback_verified']
            and staged['source_unchanged'] and staged['changed_blocks'] == 1
            and staged['image_sha256'] == TARGET_SHA and staged['source_sha256'] == SOURCE_SHA
            and staged['bytes'] == SIZE, 'Staged image receipt mismatch')
    audit = load(before / 'audit.json')
    receipt = load(before / 'local-verification.json')
    require(audit['complete'] and receipt['complete'] and receipt['both_macs_verified'],
            'Fresh before capture not verified on both Macs')
    require(receipt['files'] == audit['files'], 'Before receipt file mapping mismatch')
    for name, (start, count) in BOUNDS.items():
        row = audit['files'][name]
        require(row['file'] == name + '.bin' and row['bytes'] == count * 512
                and row['sha256'] == PINS[name] and row['start_sector'] == start
                and row['sectors'] == count and row['final_ack'] is True,
                'Wrong before capture: ' + name)
        checked_file(before / row['file'], count * 512, PINS[name])
    primary = (before / 'gpt.bin').read_bytes()
    gpt = read_gpt(primary)
    for name in ('system', 'boot', 'recovery', 'misc'):
        part = gpt['partitions'][name]
        require((part['start_sector'], part['sectors']) == BOUNDS[name], 'GPT bounds mismatch: ' + name)
    require(not any((before / 'misc.bin').read_bytes()[:32]), 'BCB command present; separate review needed')
    with (before / 'system.bin').open('rb') as source, (package / 'system.img').open('rb') as target:
        for offset, original, block in blocks:
            source.seek(offset)
            target.seek(offset)
            require(sha(source.read(4096)) == original and target.read(4096) == block,
                    'Delta differs from saved source/target')
    return blocks, audit, receipt


class SystemDeltaWriter(Reader):
    def set_write_mode(self, enabled):
        self.send('configure', {'MemoryName': 'eMMC', 'Verbose': 0,
                               'AlwaysValidate': 0, 'MaxDigestTableSizeInBytes': 2048,
                               'MaxPayloadSizeToTargetInBytes': 65536,
                               'ZLPAwareHost': 1, 'SkipStorageInit': 0,
                               'SkipWrite': int(not enabled)})
        attrs = self.response()
        require(attrs.get('MemoryName', 'eMMC').lower() == 'emmc', 'Wrong storage')
        limit = int(attrs.get('MaxPayloadSizeToTargetInBytes', 65536))
        self.chunk_size = min(65536, limit)
        require(self.chunk_size >= 512 and self.chunk_size % 512 == 0, 'Bad payload limit')
        self.writes_enabled = enabled

    def program_block(self, offset, block):
        require(getattr(self, 'writes_enabled', False), 'Writes are disabled')
        require(type(offset) is int and offset in OFFSETS and len(block) == 4096,
                'Only the one reviewed full blocks may be programmed')
        require(self.allowed_blocks.get(offset) == block, 'Payload is not the exact reviewed block')
        root = ET.Element('data')
        ET.SubElement(root, 'program', {'SECTOR_SIZE_IN_BYTES': '512',
            'physical_partition_number': '0', 'start_sector': str(START + offset // 512),
            'num_partition_sectors': '8'})
        encoded = b'<?xml version="1.0" ?>' + ET.tostring(root, encoding='utf-8')
        self.log.write(b'TX ' + encoded + b'\n')
        self.log.flush()
        if self.ep_out.write(encoded, timeout=3000) != len(encoded):
            raise RuntimeError('Short program XML; do not retry')
        self.response(rawmode=True)
        for at in range(0, len(block), self.chunk_size):
            chunk = block[at:at + self.chunk_size]
            if self.ep_out.write(chunk, timeout=10000) != len(chunk):
                raise RuntimeError('Short raw program transfer; keep EDL, do not retry/reset')
            if self.ep_out.write(b'', timeout=3000) != 0:
                raise RuntimeError('Unexpected ZLP result; keep EDL, do not retry/reset')
        self.response(rawmode=False, timeout=30)


def verify_small(client, directory):
    directory.mkdir(mode=0o700, exist_ok=False)
    result = {}
    for name in ('gpt', 'boot', 'recovery', 'misc'):
        start, count = BOUNDS[name]
        path = directory / (name + '.bin')
        row = client.read(start, count, path)
        require(row['sha256'] == PINS[name], 'Live protected partition differs: ' + name)
        checked_file(path, count * 512, PINS[name])
        result[name] = row
    return result


def execute(blocks, before, audit, receipt):
    review = load(ROOT / 'test23-edl-system-independent-review.json')
    require(review['passed'] and review['runner_sha256'] == file_sha(Path(__file__))
            and review['delta_sha256'] == DELTA_SHA, 'Independent runner gate missing/mismatched')
    require(shutil.disk_usage(ROOT).free > SIZE + 512 * 1024 * 1024,
            'Insufficient space for full after readback plus protected partition captures')
    out = ROOT / 'test23-edl-system-write'
    out.mkdir(mode=0o700, exist_ok=False)
    state_path = out / 'transaction.json'
    state = {'partition': 'system', 'start_sector': START, 'sectors': SECTORS,
             'source_sha256': SOURCE_SHA, 'target_sha256': TARGET_SHA,
             'delta_sha256': DELTA_SHA, 'before_audit_sha256': file_sha(before / 'audit.json'),
             'before_receipt_sha256': file_sha(before / 'local-verification.json'),
             'write_started': False, 'blocks_completed': [], 'in_progress_offset': None,
             'full_readback_verified': False, 'protected_partitions_verified': False,
             'reset_sent': False, 'userdata_written': False, 'automatic_retry': False}
    save(state_path, state)
    def deadline(*unused):
        raise TimeoutError('30-minute EDL transaction deadline; keep EDL, no retry/reset')
    signal.signal(signal.SIGALRM, deadline)
    signal.alarm(1800)
    try:
        with (out / 'firehose.xml.log').open('xb') as log:
            client = SystemDeltaWriter(log)
            try:
                client.allowed_blocks = {offset: block for offset, _, block in blocks}
                client.configure()
                state['protected_before'] = verify_small(client, out / 'protected-before')
                originals = out / 'original-blocks'
                originals.mkdir(mode=0o700)
                for offset, original, block in blocks:
                    path = originals / (str(offset) + '.bin')
                    row = client.read(START + offset // 512, 8, path)
                    require(row['sha256'] == original, 'Current block differs from exact Test22')
                    checked_file(path, 4096, original)
                state['live_preflight_verified'] = True
                save(state_path, state)
                readbacks = out / 'block-readbacks'
                readbacks.mkdir(mode=0o700)
                for offset, original, block in blocks:
                    state.update(write_started=True, in_progress_offset=offset)
                    save(state_path, state)
                    client.set_write_mode(True)
                    client.program_block(offset, block)
                    client.set_write_mode(False)
                    path = readbacks / (str(offset) + '.bin')
                    row = client.read(START + offset // 512, 8, path)
                    require(row['sha256'] == sha(block) and path.read_bytes() == block,
                            'Block readback mismatch; keep EDL, do not retry/reset')
                    state['blocks_completed'].append(offset)
                    state['in_progress_offset'] = None
                    save(state_path, state)
                    print(json.dumps({'verified_blocks': len(state['blocks_completed']), 'total_blocks': 1}), flush=True)
                after = out / 'system-after.bin'
                state['system_after'] = client.read(START, SECTORS, after)
                require(state['system_after']['sha256'] == TARGET_SHA, 'FULL SYSTEM READBACK MISMATCH')
                checked_file(after, SIZE, TARGET_SHA)
                state['full_readback_verified'] = True
                save(state_path, state)
                state['protected_after'] = verify_small(client, out / 'protected-after')
                state.update(protected_partitions_verified=True, complete=True)
                save(state_path, state)
                print(json.dumps({'complete': True, 'full_system_sha256': TARGET_SHA,
                                  'verified_blocks': 1, 'reset_sent': False}), flush=True)
            finally:
                client.close()
    except BaseException as exc:
        state['error'] = type(exc).__name__ + ': ' + str(exc)
        save(state_path, state)
        raise
    finally:
        signal.alarm(0)


def main():
    os.umask(0o077)
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--package', type=Path, default=ROOT / 'test23-package-v1')
    parser.add_argument('--before', type=Path, default=ROOT / 'test23-edl-before')
    parser.add_argument('--execute', action='store_true')
    args = parser.parse_args()
    blocks, audit, receipt = preflight(args.package, args.before)
    print(json.dumps({'files_only_preflight_passed': True, 'blocks': len(blocks),
                      'write_bytes': len(blocks) * 4096, 'execute': args.execute}), flush=True)
    if args.execute:
        execute(blocks, args.before, audit, receipt)


if __name__ == '__main__':
    main()

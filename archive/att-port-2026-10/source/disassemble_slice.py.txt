#!/usr/bin/env python3
"""Wrap a private raw arm64 kernel range in a temporary ELF for llvm-objdump."""
from pathlib import Path
import struct,sys,tempfile,subprocess
base=0xffffff8008080000
start,end=map(lambda s:int(s,0),sys.argv[1:3])
raw=(Path(__file__).parent/'kernel-analysis/twrp-3.7.0.Image').read_bytes()[start-base:end-base]
assert len(raw)==end-start
names=b'\0.text\0.shstrtab\0'
off=64;strings=off+len(raw);shoff=(strings+len(names)+7)&~7
header=struct.pack('<16sHHIQQQIHHHHHH',b'\x7fELF\x02\x01\x01'+b'\0'*9,1,183,1,0,0,shoff,0,64,0,0,64,3,2)
section=lambda *args:struct.pack('<IIQQQQIIQQ',*args)
data=header+raw+names+b'\0'*(shoff-strings-len(names))+b'\0'*64
data+=section(1,1,6,start,off,len(raw),0,0,4,0)
data+=section(7,3,0,0,strings,len(names),0,0,1,0)
with tempfile.NamedTemporaryFile(suffix='.elf') as f:
    f.write(data);f.flush()
    subprocess.run(['/usr/bin/objdump','-d',f.name],check=True)

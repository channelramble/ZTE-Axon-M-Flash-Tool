#!/usr/bin/env python3
"""Add a slot-identification file to the hardware-proven 3.18 reader."""
import gzip
import hashlib
import json
import os
from pathlib import Path
import struct
import subprocess
import sys

sys.dont_write_bytecode = True
ROOT = Path(__file__).resolve().parent
sys.path.insert(0, str(ROOT.parent / "ZTE-Axon-M-Flash-Tool/scripts"))
from fujisan_image import BootImage, cpio_entries, gzip_member, report, sign_image

SOURCE_SHA = "76990c6f20c61a919629600c9afbb4551e0792e911f05034927852c9fcdefe28"
MARKER_NAME = "fujisan-reader-slot"
MARKER_CONTENT = b"recovery-control-20261016\n"


def sha(data):
    return hashlib.sha256(data).hexdigest()


def require(condition, message):
    if not condition:
        raise ValueError(message)


def add_marker(ramdisk):
    _, raw, tail = gzip_member(ramdisk)
    require(not tail, "Unexpected ramdisk gzip trailer")
    before = cpio_entries(raw)
    require(before[-1][0] == "TRAILER!!!", "Missing CPIO trailer")
    require(all(entry[0].removeprefix("./") != MARKER_NAME for entry in before),
            "Source already contains the slot marker")
    inode = max(entry[1][0] for entry in before) + 1
    require(inode <= 0xffffffff, "No unused CPIO inode available")
    filename = MARKER_NAME.encode("ascii") + b"\0"
    fields = [inode, 0o100444, 0, 0, 1, 0, len(MARKER_CONTENT),
              0, 0, 0, 0, len(filename), 0]
    entry = b"070701" + "".join(f"{value:08x}" for value in fields).encode("ascii")
    entry += filename
    entry += b"\0" * (-len(entry) % 4)
    entry += MARKER_CONTENT
    entry += b"\0" * (-len(entry) % 4)
    rebuilt = b"".join(item[3] for item in before[:-1]) + entry + before[-1][3]
    rebuilt += b"\0" * (-len(rebuilt) % 512)
    after = cpio_entries(rebuilt)
    require(len(after) == len(before) + 1, "Unexpected CPIO entry count")
    require(after[-2] == (MARKER_NAME, fields, MARKER_CONTENT, entry),
            "Marker entry roundtrip mismatch")
    require(after[:-2] + after[-1:] == before,
            "An existing encoded CPIO entry changed")
    external = subprocess.run(["/usr/bin/cpio", "-it"], input=rebuilt,
                              capture_output=True, check=False)
    require(external.returncode == 0, "External cpio rejected rebuilt archive")
    require(MARKER_NAME in external.stdout.decode("utf-8").splitlines(),
            "External cpio did not list marker")
    compressed = gzip.compress(rebuilt, mtime=0)
    require(gzip_member(compressed)[1] == rebuilt, "Ramdisk gzip roundtrip failed")
    return compressed, before, after, sha(raw), sha(rebuilt)


def main():
    os.umask(0o077)
    source = ROOT / "prepared/reader-3.2.3-boot.img"
    target = ROOT / "prepared/reader-3.2.3-recovery-control.img"
    validation_path = target.with_suffix(".validation.json")
    inspection_path = target.with_suffix(".img.json")
    require(not any(path.exists() for path in (target, validation_path, inspection_path)),
            "Refusing to overwrite an existing control artifact")
    original = source.read_bytes()
    require(sha(original) == SOURCE_SHA, "Source is not the hardware-proven test06 reader")
    boot = BootImage(original)
    source_info = report(original)
    require(source_info["signature_verified"] and source_info["boot_id_valid"]
            and source_info["signature_target"] == "/boot", "Source validation failed")
    ramdisk, before, after, old_cpio_sha, new_cpio_sha = add_marker(boot.ramdisk)
    signed = sign_image(boot.rebuild(ramdisk=ramdisk), "/recovery",
                        str(ROOT / "keys/axon-m-local.key"),
                        str(ROOT / "keys/axon-m-local.pem"))
    result = BootImage(signed)
    require(result.kernel == boot.kernel and result.second == boot.second,
            "Kernel, appended DTBs or second section changed")
    require(result.cmdline == boot.cmdline and result.page == boot.page,
            "Command line or page size changed")
    header_before, header_after = bytearray(boot.header), bytearray(result.header)
    for start, end in ((16, 20), (576, 608)):
        header_before[start:end] = b"\0" * (end - start)
        header_after[start:end] = b"\0" * (end - start)
    require(header_before == header_after,
            "Boot header changed outside ramdisk size and boot ID")
    require(cpio_entries(gzip_member(result.ramdisk)[1]) == after,
            "Final ramdisk differs from validated archive")
    info = report(signed)
    require(info["signature_verified"] and info["boot_id_valid"]
            and info["signature_target"] == "/recovery", "Final image validation failed")
    validation = {
        "source_image": source.name,
        "source_sha256": SOURCE_SHA,
        "image": target.name,
        "image_bytes": len(signed),
        "image_sha256": sha(signed),
        "padded_bytes": len(signed) + (-len(signed) % 512),
        "padded_sha256": sha(signed + b"\0" * (-len(signed) % 512)),
        "marker_path": "/" + MARKER_NAME,
        "marker_contents": MARKER_CONTENT.decode("ascii"),
        "marker_mode": "0100444",
        "marker_uid": 0,
        "marker_gid": 0,
        "marker_links": 1,
        "cpio_entries_before": len(before),
        "cpio_entries_after": len(after),
        "all_existing_encoded_cpio_entries_unchanged": True,
        "cpio_sha256_before": old_cpio_sha,
        "cpio_sha256_after": new_cpio_sha,
        "external_cpio_list_passed": True,
        "kernel_and_appended_dtbs_unchanged": True,
        "kernel_sha256": sha(result.kernel),
        "init_rc_unchanged": True,
        "cmdline_unchanged": True,
        "cmdline_bytes": len(result.cmdline.encode("ascii")),
        "header_changed_only_ramdisk_size_and_id": True,
        "kernel_address": hex(struct.unpack_from("<I", result.header, 12)[0]),
        "ramdisk_address": hex(struct.unpack_from("<I", result.header, 20)[0]),
        "tags_address": hex(struct.unpack_from("<I", result.header, 32)[0]),
        "signature_target": "/recovery",
        "signature_verified": True,
        "boot_id_valid": True,
        "hardware_tested": False,
        "interpretation": "The marker proves this recovery ramdisk executed; TWRP without it may be fallback to the unchanged boot reader. No change to kernel, DT, init actions or reboot behavior."
    }
    with target.open("xb") as handle:
        handle.write(signed)
    for path, contents in ((validation_path, validation), (inspection_path, info)):
        with path.open("x") as handle:
            json.dump(contents, handle, indent=2)
            handle.write("\n")
    print(json.dumps(validation, indent=2))


if __name__ == "__main__":
    main()

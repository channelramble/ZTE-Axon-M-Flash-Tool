#!/usr/bin/env python3
"""Keep crypto initialization, suppress only its two optional QCE devices."""
import hashlib
import json
import os
from pathlib import Path
import sys

ROOT = Path(__file__).resolve().parent
sys.path.insert(0, str(ROOT.parent/'ZTE-Axon-M-Flash-Tool/scripts'))
from fujisan_image import BootImage, Fdt, kernel_dtbs, report, sign_image
from prepare_test04_led import flatten

SOURCE_SHA = '6a80d22902a4545eb90a9ee610ba7252dde4d50ee1929679497666ded2425739'
NODES = {'qcrypto@660000': b'qcom,qcrypto\0', 'qcedev@660000': b'qcom,qcedev\0'}


def sha(data):
    return hashlib.sha256(data).hexdigest()


def main():
    os.umask(0o077)
    source=ROOT/'prepared/test-04-blue-led-boot.img'
    target=ROOT/'prepared/test-05-qce-disabled-boot.img'
    data=source.read_bytes()
    if sha(data) != SOURCE_SHA:
        raise ValueError('Source differs from verified test04')
    boot=BootImage(data)
    compressed, raw, trees=kernel_dtbs(boot.kernel)
    if len(trees) != 2:
        raise ValueError('Expected both published device trees')
    blobs=[];changes=[]
    for index,(old_blob,tree) in enumerate(trees):
        before=flatten(tree.root)
        soc=tree.root.child('soc')
        for name,compatible in NODES.items():
            node=soc.child(name)
            if node is None or node.props.get('compatible') != compatible:
                raise ValueError('Unexpected QCE node/compatible')
            if node.props.get('reg') != bytes.fromhex('00660000000200000064400000024000'):
                raise ValueError('Unexpected QCE hardware ranges')
            if 'status' in node.props:
                raise ValueError('Expected initially enabled QCE node without status property')
            node.props['status']=b'disabled\0'
        new_blob=tree.serialize(free=8192)
        after=flatten(Fdt(new_blob).root)
        changed={key for key in before.keys()|after.keys() if before.get(key)!=after.get(key)}
        expected={('//soc/'+name,'status') for name in NODES}
        if changed != expected:
            raise ValueError('Unexpected semantic DT changes: '+str(changed))
        # The full-property comparison also proves eMMC/ICE/USB/LED unchanged.
        blobs.append(new_blob)
        changes.append({'dt_index':index,'nodes':['/soc/'+name for name in NODES],
                        'property':'status','before':'absent (enabled)','after':'disabled',
                        'original_sha256':sha(old_blob),'sha256':sha(new_blob)})
    signed=sign_image(boot.rebuild(kernel=compressed+b''.join(blobs)),'/boot',
                      str(ROOT/'keys/axon-m-local.key'),str(ROOT/'keys/axon-m-local.pem'))
    rebuilt=BootImage(signed);comp2,raw2,trees2=kernel_dtbs(rebuilt.kernel)
    if comp2!=compressed or raw2!=raw or [b for b,_ in trees2]!=blobs:
        raise ValueError('Kernel/DT roundtrip mismatch')
    if rebuilt.ramdisk!=boot.ramdisk or rebuilt.cmdline!=boot.cmdline or rebuilt.second!=boot.second:
        raise ValueError('Unexpected non-DT image change')
    info=report(signed)
    if not info['signature_verified'] or not info['boot_id_valid'] or any(x['free_tail_bytes']!=8192 for x in info['dtbs']):
        raise ValueError('Final validation failed')
    v=dict(source_image=source.name,source_sha256=SOURCE_SHA,image=target.name,
           image_bytes=len(signed),image_sha256=sha(signed),
           padded_sha256=sha(signed+b'\0'*(-len(signed)%512)),kernel_raw_sha256=sha(raw),
           kernel_gzip_unchanged=True,ramdisk_unchanged=True,cmdline_unchanged=True,
           cmdline_bytes=len(boot.cmdline.encode()),dt_changes=changes,
           signature_verified=True,boot_id_valid=True,hardware_tested=False,
           shared_crypto_initializers_retained=True,blue_marker_retained=True,
           interpretation='Tests optional QCE clock/MMIO/BAM probes before the LED marker. Software crypto remains available; Qualcomm-specific encrypted-storage operations may fail. A boot or fresh blue transition is informative; continued red is inconclusive.')
    with target.open('xb') as f:f.write(signed)
    with target.with_suffix('.validation.json').open('x') as f:json.dump(v,f,indent=2);f.write('\n')
    with target.with_suffix('.img.json').open('x') as f:json.dump(info,f,indent=2);f.write('\n')
    print(json.dumps(v,indent=2))


if __name__=='__main__':
    main()

#!/usr/bin/env python3
"""Local-only, guarded LED pulse diagnostic; preserve deferred probing itself."""
import gzip, hashlib, json, os, struct, subprocess, sys, tempfile
from pathlib import Path

ROOT=Path(__file__).resolve().parent
sys.path.insert(0,str(ROOT.parent/'ZTE-Axon-M-Flash-Tool/scripts'))
from fujisan_image import BootImage,gzip_member,report,sign_image
SHA=lambda b:hashlib.sha256(b).hexdigest()
BASE=0xffffff8008080000
CAVE=0xffffff8009786674
CAVE_END=0xffffff800978684c
EXTERNAL={'msleep':0xffffff8008128a9c,'down_read':0xffffff800904a9f4,
          'strcmp':0xffffff800902fdc0,'__qpnp_led_work':0xffffff8008a241bc,
          'up_read':0xffffff800810861c}

def branch(pc,target):
    delta=target-pc
    if delta%4 or not -(1<<27)<=delta<(1<<27):
        raise ValueError('BL out of range')
    return struct.pack('<I',0x94000000|((delta//4)&0x3ffffff))

def link_object(data):
    if data[:7]!=b'\x7fELF\x02\x01\x01' or struct.unpack_from('<HH',data,16)!=(1,183):
        raise ValueError('Expected little-endian arm64 relocatable ELF')
    shoff=struct.unpack_from('<Q',data,40)[0]
    entsize,count,strindex=struct.unpack_from('<HHH',data,58)
    assert entsize==64
    sections=[struct.unpack_from('<IIQQQQIIQQ',data,shoff+i*64) for i in range(count)]
    content=lambda s:data[s[4]:s[4]+s[5]]
    strings=content(sections[strindex])
    cstr=lambda table,pos:table[pos:table.index(b'\0',pos)].decode()
    named={cstr(strings,s[0]):(i,s) for i,s in enumerate(sections)}
    textindex,textsection=named['.text'];code=bytearray(content(textsection))
    syms=content(named['.symtab'][1]);names=content(sections[named['.symtab'][1][6]])
    symbols=[]
    for off in range(0,len(syms),24):
        name,info,other,shndx,value,size=struct.unpack_from('<IBBHQQ',syms,off)
        symbols.append((cstr(names,name),shndx,value))
    addresses={name:CAVE+value for name,idx,value in symbols if idx==textindex and not name.startswith('$')}
    assert addresses['_start']==CAVE
    assert set(addresses)=={'_start','pre_enable_marker','post_flush_marker','green_sync'}
    calls=[]
    for name,(i,s) in named.items():
        if s[1] not in (4,9):continue
        assert name=='.rela.text' and s[1]==4 and s[7]==textindex and s[9]==24
        rel=content(s)
        for off in range(0,len(rel),24):
            offset,info,addend=struct.unpack_from('<QQq',rel,off)
            assert info&0xffffffff==283 and addend==0
            symbol,idx,value=symbols[info>>32]
            target=addresses[symbol] if idx==textindex else EXTERNAL[symbol]
            assert code[offset:offset+4]==bytes.fromhex('00000094')
            code[offset:offset+4]=branch(CAVE+offset,target)
            calls.append(dict(offset=hex(offset),symbol=symbol,target=hex(target)))
    assert len(calls)==10
    return bytes(code),addresses,calls

def main():
    os.umask(0o077)
    source=ROOT/'prepared/test-09-end-level6-green-boot-adb-only.img'
    target=ROOT/'prepared/test-11-deferred-pulse-v3-boot-adb-only.img'
    paths=[target,target.with_suffix('.validation.json'),target.with_suffix('.img.json')]
    if any(p.exists() for p in paths):raise FileExistsError('Immutable test artifact exists')
    data=source.read_bytes()
    assert SHA(data)=='fd9640cdb44b3c5d9081cb91129020e5d2afbe619fdd21b9c33385d055d80568'
    boot=BootImage(data);_,raw,tail=gzip_member(boot.kernel)
    assert SHA(raw)=='3ac4beba7de46c1a10e89ce42adcb3dc46d0d987fb48d009917301ddea9ee107'
    smap=(ROOT/'kernel-analysis/twrp-3.7.0.System.map').read_bytes()
    assert SHA(smap)=='532606e92b85d81d98786af6263d03747bf2644705fb794df5614b96add386d0'
    mapping={line.split()[-1]:int(line.split()[0],16) for line in smap.decode().splitlines() if len(line.split())==3}
    for name,address in EXTERNAL.items():assert mapping[name]==address
    assert mapping['msm_pil_init']==CAVE and mapping['msm_performance_init']==CAVE_END
    blacklist=next(x.split('=',1)[1].split(',') for x in boot.cmdline.split() if x.startswith('initcall_blacklist='))
    assert len(blacklist)==8 and 'msm_pil_init' in blacklist
    assert SHA(raw[CAVE-BASE:CAVE_END-BASE])=='f0aedb3446e95a23322ad53f6af6f0eca2ad629752486e543d37831ee4bef469'
    # No direct B/BL may enter the repurposed body from outside it.
    for offset in range(0,len(raw)-3,4):
        word=struct.unpack_from('<I',raw,offset)[0]
        if word&0x7c000000==0x14000000:
            imm=word&0x3ffffff;imm=imm-(1<<26) if imm&(1<<25) else imm
            dest=BASE+offset+4*imm
            assert not (CAVE<=dest<CAVE_END and not CAVE<=BASE+offset<CAVE_END)
    asm=ROOT/'test11_deferred_marker_v3.S'
    assert SHA(asm.read_bytes())=='b1249aaba2276d05f537d27b265a2ddd3e699ba800a1f8b0d39043955f86804a'
    with tempfile.TemporaryDirectory() as d:
        obj=Path(d)/'marker.o'
        subprocess.run(['/usr/bin/clang','--target=aarch64-none-elf','-c',str(asm),'-o',str(obj)],check=True)
        code,addresses,calls=link_object(obj.read_bytes())
    assert len(code)<=CAVE_END-CAVE
    patched=bytearray(raw);patched[CAVE-BASE:CAVE-BASE+len(code)]=code
    hooks=[(0xffffff800867ec48,bytes.fromhex('28008052'),'pre_enable_marker'),
           (0xffffff800867ecd8,bytes.fromhex('e0031f2a'),'post_flush_marker')]
    for pc,expected,name in hooks:
        assert raw[pc-BASE:pc-BASE+4]==expected
        patched[pc-BASE:pc-BASE+4]=branch(pc,addresses[name])
    # The original trigger, queue and flush calls, and all 690 table entries stay exact.
    assert patched[0x5fec4c:0x5fecd8]==raw[0x5fec4c:0x5fecd8]
    assert patched[0x174be28:0x174d3b8]==raw[0x174be28:0x174d3b8]
    changes=[i for i,(a,b) in enumerate(zip(raw,patched)) if a!=b]
    assert all(CAVE-BASE<=i<CAVE-BASE+len(code) or any(pc-BASE<=i<pc-BASE+4 for pc,_,_ in hooks) for i in changes)
    signed=sign_image(boot.rebuild(kernel=gzip.compress(bytes(patched),compresslevel=9,mtime=0)+tail),
                      '/boot',str(ROOT/'keys/axon-m-local.key'),str(ROOT/'keys/axon-m-local.pem'))
    new=BootImage(signed);_,checked,newtail=gzip_member(new.kernel)
    assert checked==patched and tail==newtail and new.ramdisk==boot.ramdisk and new.cmdline==boot.cmdline
    assert boot.rebuild(kernel=new.kernel)==new.payload
    info=report(signed)
    assert info['signature_verified'] and info['boot_id_valid'] and info['signature_target']=='/boot'
    assert all(t['free_tail_bytes']==8192 for t in info['dtbs'])
    v=dict(source_image=source.name,source_sha256=SHA(data),image=target.name,image_bytes=len(signed),
           image_sha256=SHA(signed),padded_sha256=SHA(signed+b'\0'*(-len(signed)%512)),
           raw_sha256=SHA(patched),helper_sha256=SHA(code),helper_bytes=len(code),
           helper_addresses={n:hex(a) for n,a in addresses.items()},helper_calls=calls,
           raw_changed_bytes=len(changes),raw_changed_offsets=changes,
           hooks=[dict(address=hex(pc),original=orig.hex(),replacement=branch(pc,addresses[n]).hex(),helper=n) for pc,orig,n in hooks],
           all_690_initcalls_identical_to_test09=True,deferred_queue_flush_unchanged=True,
           dt_tail_unchanged=True,ramdisk_unchanged=True,cmdline_unchanged=True,
           signature_verified=True,boot_id_valid=True,hardware_tested=False,
           interpretation='Retain Test09 registration before sync. Before enabling deferred retries, matched green is synchronously off500ms/on500ms/off. After original flush returns, matched green is on. Complete observed pulse then red implicates the enabled cycle or instrumentation afterward; pulse then stable green supports flush return. No pulse is inconclusive. No LED registration repetition, new skip, or deferred work suppression. Timing changes by1s when green is found.')
    for path,content in [(paths[0],signed),(paths[1],(json.dumps(v,indent=2)+'\n').encode()),(paths[2],(json.dumps(info,indent=2)+'\n').encode())]:
        with path.open('xb') as f:f.write(content)
    (ROOT/'test11-v3-linked-helper.bin').write_bytes(code)
    print(json.dumps({k:x for k,x in v.items() if k!='raw_changed_offsets'},indent=2))

if __name__=='__main__':main()
